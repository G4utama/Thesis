\newglossaryentry{Annotazioni}{
    name={Annotazioni},
    description={nel linguaggio di programmazione \nameref{Java}, un'annotazione è una forma di metadati sintattici che può essere aggiunta al codice sorgente \nameref{Java}, proprio come un attributo. È possibile annotare classi, metodi, variabili, parametri e pacchetti \nameref{Java}}    
}

\newglossaryentry{Bean}{
    name={Bean},
    description={classi che incapsulano uno o più oggetti in un unico oggetto standardizzato (il \gls{Bean}). Questa standardizzazione consente di gestire i \gls{Bean} in modo più generico, facilitando il riutilizzo del codice e l'introspezione. Ciò permette, a sua volta, di trattare i \gls{Bean} come componenti \textit{software} e di manipolarli visivamente tramite \textit{editor} e \acrshort{IDE} senza bisogno di alcuna configurazione iniziale né di conoscere alcun dettaglio di implementazione interna.}  
}

\newglossaryentry{Checkpointing}{
    name={Checkpointing},
    description={tecnica che garantisce la tolleranza ai guasti nei sistemi informatici. Consiste nel salvare un'istantanea dello stato di un'applicazione, in modo che questa possa riavviarsi da quel punto in caso di guasto. Ciò è particolarmente importante per le applicazioni a esecuzione prolungata che vengono eseguite in sistemi informatici soggetti a guasti}  
}

\newglossaryentry{Dependency Injection}{
    name={Dependency Injection},
    description={tecnica di programmazione in cui un oggetto o una funzione riceve altri oggetti o funzioni di cui ha bisogno, anziché crearli internamente. La \textit{Dependency Injection} mira a separare gli aspetti relativi alla creazione degli oggetti da quelli relativi al loro utilizzo, portando a programmi con un accoppiamento debole. Questo modello garantisce che un oggetto o una funzione che desideri utilizzare un determinato servizio non debba sapere come creare tali servizi. Al contrario, al "cliente" ricevente (oggetto o funzione) vengono fornite le sue dipendenze da un codice esterno (un "iniettore"), di cui esso non è a conoscenza}
}

\newglossaryentry{Framework}{
    name={Framework},
    description={insieme strutturato di librerie, strumenti e convenzioni che fornisce un'infrastruttura di base su cui costruire un'applicazione \textit{software}. A differenza di una singola libreria, che viene invocata dal codice dello sviluppatore, un \textit{\gls{Framework}} definisce lo scheletro dell'applicazione e ne richiama a sua volta il codice specifico (\gls{Inversion of Control}), occupandosi degli aspetti infrastrutturali comuni e lasciando allo sviluppatore la sola logica di dominio}
}

\newglossaryentry{Hibernate}{
    name={Hibernate},
    description={strumento di mappatura oggetto-relazionale per il linguaggio di programmazione \nameref{Java}. Fornisce un \textit{\gls{Framework}} per la mappatura di un modello di dominio orientato agli oggetti su un database relazionale. Hibernate risolve i problemi di disallineamento oggetto-relazionale sostituendo gli accessi diretti e persistenti al database con funzioni di gestione degli oggetti di alto livello}  
}

\newglossaryentry{Inversion of Control}{
    name={Inversion of Control},
    description={espressione adottata separatamente dalla comunità dei programmatori \nameref{Java} per riferirsi specificatamente ai modelli di \textit{\gls{Dependency Injection}} (il passaggio di servizi agli oggetti che ne hanno bisogno) che si verificano con i \textit{container} \acrshort{IoC} nei \textit{\gls{Framework}} \nameref{Java} come \nameref{Spring}. In questo senso, si riferisce al fatto di concedere al \textit{\gls{Framework}} il controllo sulle implementazioni delle dipendenze utilizzate dagli oggetti dell'applicazione, piuttosto che al significato originario di concedere al \textit{\gls{Framework}} il flusso di controllo}  
}

\newglossaryentry{Jakarta Persistence}{
    name={Jakarta Persistence},
    description={specifica \nameref{Java} che definisce un insieme \textit{standard} di annotazioni e interfacce per la mappatura oggetto-relazionale, implementata concretamente da strumenti come \gls{Hibernate}}
}

\newglossaryentry{Kafka Sink}{
    name={Kafka Sink},
    description={connettore nativo offerto da Apache Flink per la scrittura di un flusso di eventi (\textit{stream}) su un \gls{Topic Kafka} di destinazione. Rappresenta il punto di uscita di una \textit{pipeline} Flink verso Kafka, complementare al \gls{Kafka Source}, e può essere configurato con garanzie di consegna diverse (\textit{At-least-once} o \textit{Exactly-once}), quest'ultima ottenibile in combinazione con il meccanismo di \textit{\gls{Checkpointing}} della piattaforma}
}

\newglossaryentry{Kafka Source}{
    name={Kafka Source},
    description={connettore nativo offerto da Apache Flink per la lettura di un \gls{Topic Kafka} come flusso continuo di eventi (\textit{stream}). Si occupa della deserializzazione dei singoli messaggi nel formato atteso dall'applicazione e della gestione dell'\textit{offset} di lettura, rappresentando il punto di ingresso di una \textit{pipeline} Flink alimentata da Kafka}
}

\newglossaryentry{KRaft}{
    name={KRaft},
    description={abbreviazione di Kafka Raft, l'architettura di consenso distribuito adottata dalle versioni più recenti di \nameref{Apache Kafka} per il coordinamento del \textit{cluster} (elezione del \textit{broker} controllore, gestione dei metadati), che elimina la dipendenza da Apache ZooKeeper, storicamente richiesto a questo scopo, unificando i ruoli di \textit{broker} e \textit{controller} anche in un singolo nodo}
}

\newglossaryentry{Lombok}{
    name={Lombok},
    description={libreria \nameref{Java} che si integra automaticamente con l'\textit{editor} e gli strumenti di compilazione, arricchendo il codice \nameref{Java}.}  
}

\newglossaryentry{Open Source}{
    name={Open Source},
    description={software distribuito, generalmente in via gratuita, sotto i termini di una licenza \textit{\gls{Open Source}}}
}

\newglossaryentry{Side Output}{
    name={Side Output},
    description={meccanismo offerto da Apache Flink che consente a un operatore di una \textit{pipeline} di emettere, oltre al flusso di uscita principale, uno o più flussi di uscita secondari (\textit{side output}), tipicamente utilizzati per instradare eventi che non possono essere elaborati correttamente verso un canale separato, senza interrompere l'elaborazione del flusso principale}
}

\newglossaryentry{Throughput}{
    name={Throughput},
    description={La velocità di trasmissione (Throughput) indica la velocità di consegna dei messaggi su un canale di comunicazione all'interno di una rete di comunicazione. 
    I dati contenuti in questi messaggi possono essere trasmessi tramite collegamenti fisici o logici, oppure attraverso nodi di rete. La velocità di trasmissione rappresenta il consumo di larghezza di banda digitale. 
    La velocità effettiva di un sistema di comunicazione può essere influenzata da vari fattori, tra cui i limiti del mezzo fisico sottostante, la potenza di elaborazione disponibile dei componenti del sistema, il comportamento dell'utente finale}
}

\newglossaryentry{Topic Kafka}{
    name={Topic Kafka},
    description={categorie utilizzate per organizzare i messaggi. Ogni \textit{\nameref{Topic}} ha un nome unico per l'intero \textit{cluster} Kafka.
    I messaggi vengono inviati e letti da argomenti specifici. I produttori scrivono dati negli argomenti e i consumatori leggono dati dagli argomenti. 
    In Kafka, gli argomenti sono partizionati e replicati attraverso i \textit{broker} per tutta l'implementazione.  I \textit{broker} si riferiscono a ciascuno dei nodi di un \textit{cluster} Kafka. Le partizioni sono importanti perché permettono la parallelizzazione degli argomenti, consentendo un elevato \textit{\gls{Throughput}} dei messaggi}
}
