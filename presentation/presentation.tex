\begin{frame}{Perché serve un Middleware?}
    \begin{columns}[c]
        \column{0.5\textwidth}
        \textbf{Problema}:
        \vspace{6pt}
        \begin{itemize}
            \setlength\itemsep{12pt}
            \item \textbf{Connessioni} dirette con ogni sistema
            \item Ognuno deve \textbf{conoscere} il formato dati dell'altro
        \end{itemize}

        \includegraphics[width=0.30\textwidth]{../assets/MiddlewareSenza.drawio.png} \\

        \vspace{6pt}
        \begin{itemize}
            \setlength\itemsep{12pt}
            \item Sistemi \textbf{eterogenei} che non comunicano direttamente
            \item \textbf{Disaccoppiamento} tra produttori e consumatori
        \end{itemize}

\begin{frame}[fragile]{Implementazione - un esempio}
    \begin{columns}[c]
        \column{0.40\textwidth}
        Jakarta Persistence:
        \vspace{6pt}
        \begin{itemize}
            \setlength\itemsep{10pt}
            \item \textbf{@Entity} 
            \item \textbf{@Table}, \textbf{@Id}, \textbf{@Column}
        \end{itemize}
        \vspace{12pt}
        Lombok:
        \vspace{6pt}
        \begin{itemize}
            \setlength\itemsep{10pt}
            \item \textbf{@Getter}, \textbf{@Setter} 
            \item \textbf{@NoArgsConstructor} 
            \item \textbf{@AllArgsConstructor} 
        \end{itemize}
        \column{0.45\textwidth}
        \footnotesize
        \begin{minted}[fontsize=\scriptsize,frame=single,framesep=10pt]{java}
@Entity
@Table(name = "customers")
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
public class Customer {
    @Id
    private Long id;
    @Column(name = "full_name", 
        nullable = false)
    private String fullName;
    @Column(name = "email")
    private String email;
    @Column(name = "city")
    private String city;
}
        \end{minted}
    \end{columns}
\end{frame}

\begin{frame}{Valutazione prestazioni}
    \begin{itemize}
        \setlength\itemsep{12pt}
        \item \textbf{Throughput}: messaggi arricchiti per secondo
        \item \textbf{Latenza end-to-end}: tempo tra evento grezzo e arricchito
        \item \textbf{Consumer lag}: quanto il sistema \textit{rincorre}
        \item \textbf{Costo interrogazione}: tempo di risposta fonte dati
    \end{itemize}
\end{frame}

% ----------
\begin{frame}{Vantaggi ottenuti}
    \begin{itemize}
        \setlength\itemsep{12pt}
        \item Sistemi \textbf{disaccoppiati}, facilmente modificabili
        \item Nessuna \textbf{perdita} di dati grazie a Kafka
        \item Errori \textbf{isolati} tramite Dead Letter Channel
        \item Ambiente \textbf{riproducibile} e facilmente replicabile
    \end{itemize}
\end{frame}
